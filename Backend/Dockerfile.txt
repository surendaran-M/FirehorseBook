# Step 1: Use a Maven image to build the project
FROM maven:3.8.5-openjdk-17 AS build

# Change this to the actual folder name in your repo
# If your code is in a folder called 'Backend', use:
COPY Backend /app
WORKDIR /app

# Run the build inside that directory
RUN mvn clean package -DskipTests

# Step 2: Use a lightweight Java image to run the project
FROM openjdk:17-jdk-slim
WORKDIR /app
# Pull the jar from the build stage
COPY --from=build /app/target/*.jar app.jar

EXPOSE 8080

ENTRYPOINT ["java","-jar","app.jar"]
